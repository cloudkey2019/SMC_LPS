# 第五章：5号位对外契约（业务规则引擎）

## 5.1 概述

5号位（业务规则引擎）是APS系统的"大脑"，负责所有业务决策逻辑。

**核心原则**：
- 5号位只做决策，不做执行
- 所有决策以凭证形式返回给2号位
- 5号位必须是无状态的，可插拔的

## 5.2 IOrderPrioritizationRule（订单打分插件）

### 5.2.1 接口定义

```csharp
/// <summary>
/// 订单优先级打分插件接口
/// 负责计算订单的优先级分数
/// </summary>
public interface IOrderPrioritizationRule
{
    /// <summary>
    /// 计算订单优先级分数
    /// </summary>
    /// <param name="orders">待打分订单列表</param>
    /// <param name="context">排产上下文</param>
    /// <returns>订单打分凭证</returns>
    OrderScoreVoucher CalculateOrderPriority(List<Order> orders, ScheduleContext context);
}
```

### 5.2.2 默认实现示例

```csharp
/// <summary>
/// 默认订单打分规则
/// 综合考虑交期、客户等级、订单金额等因素
/// </summary>
public class DefaultOrderPrioritizationRule : IOrderPrioritizationRule
{
    public OrderScoreVoucher CalculateOrderPriority(List<Order> orders, ScheduleContext context)
    {
        var voucher = new OrderScoreVoucher
        {
            PlanVersionId = context.PlanVersionId,
            DomainKey = context.DomainKey,
            CreatedBy = "DefaultOrderPrioritizationRule"
        };
        
        foreach (var order in orders)
        {
            var score = new OrderScoreDetail
            {
                OrderId = order.Id,
                OrderNo = order.OrderNo,
                ScoreBreakdown = new Dictionary<string, decimal>()
            };
            
            // 1. 交期紧急度（0-40分）
            var daysToDeadline = (order.CustomerDueDate - context.CurrentTime).TotalDays;
            var urgencyScore = daysToDeadline <= 0 ? 40m :
                              daysToDeadline <= 3 ? 35m :
                              daysToDeadline <= 7 ? 25m :
                              daysToDeadline <= 14 ? 15m : 5m;
            score.ScoreBreakdown["交期紧急度"] = urgencyScore;
            
            // 2. 基础优先级（0-30分）
            var basePriorityScore = order.Priority * 0.3m;
            score.ScoreBreakdown["基础优先级"] = basePriorityScore;
            
            // 3. 订单类型（0-20分；v5.0.24重分类：SALES_ORDER=客户订单/PRODUCTION_INSTRUCTION=生产指示）
            var typeScore = order.OrderType == "SALES_ORDER" ? 20m : 10m;
            score.ScoreBreakdown["订单类型"] = typeScore;
            
            // 4b. 客户分级加成（可选，5号位自定义实现；VIP>KEY_ACCOUNT>STANDARD>GENERAL）
            // var tierBonus = order.CustomerTier == "VIP" ? 10m : 0m;
            // score.ScoreBreakdown["客户分级"] = tierBonus;
            
            // 4c. 延迟状态加成（可选；DelayStatus独立维度，禁止与DemandMaturityStatus混用）
            // var delayBonus = order.DelayStatus == "REPEATED_DELAY" ? 15m :
            //                  order.DelayStatus == "FIRST_DELAY" ? 8m : 0m;
            // score.ScoreBreakdown["延迟状态"] = delayBonus;
            
            // 4. 库存覆盖度（0-10分）
            var coverageScore = 5m; // 简化计算
            score.ScoreBreakdown["库存覆盖度"] = coverageScore;
            
            // 计算总分
            score.PriorityScore = score.ScoreBreakdown.Values.Sum();
            score.Reason = $"交期{daysToDeadline:F0}天，优先级{order.Priority}";
            
            voucher.Scores.Add(score);
        }
        
        return voucher;
    }
}
```

### 5.3.1 接口定义

```csharp
/// 负责计算订单的库存扣减策略

public interface IPeggingRule
{
    /// <summary>
    /// 计算单个订单的Pegging策略
    /// </summary>
    /// <param name="order">订单</param>
    /// <param name="context">排产上下文</param>
    /// <returns>Pegging凭证</returns>
    PeggingVoucher CalculatePegging(Order order, ScheduleContext context);
    
    /// <summary>
    /// 扫尾孤儿订单（无库存覆盖的订单）
    /// </summary>
    /// <param name="context">排产上下文</param>
    /// <returns>孤儿订单Pegging凭证</returns>
    PeggingVoucher SweepOrphans(ScheduleContext context);
}
```

### 5.3.2 默认实现示例

```csharp
public class DefaultPeggingRule : IPeggingRule
{
    public PeggingVoucher CalculatePegging(Order order, ScheduleContext context)
    {
        var voucher = new PeggingVoucher
        {
            OrderId = order.Id,
            OrderNo = order.OrderNo,
            MaterialCode = order.MaterialCode,
            RequiredQuantity = order.Quantity,
            UOM = order.UOM,
            PlanVersionId = context.PlanVersionId,
            DomainKey = context.DomainKey,
            CreatedBy = "DefaultPeggingRule"
        };
        
        var remainingQty = order.Quantity;
        
        // 策略1：优先从库存分配
        var inventories = context.Inventories
            .Where(i => i.MaterialCode == order.MaterialCode && i.AvailableQty > 0)
            .OrderBy(i => i.Source == "ERP" ? 0 : 1) // ERP库存优先
            .ToList();
        
        foreach (var inventory in inventories)
        {
            if (remainingQty <= 0) break;
            
            var allocatedQty = Math.Min(inventory.AvailableQty, remainingQty);
            
            voucher.Details.Add(new PeggingDetail
            {
                SourceType = "INVENTORY",
                SourceId = inventory.Id,
                SourceDescription = $"{inventory.Source}库存-{inventory.MaterialCode}",
                AllocatedQuantity = allocatedQty,
                BalanceBefore = inventory.AvailableQty,
                BalanceAfter = inventory.AvailableQty - allocatedQty,
                AvailableTime = context.CurrentTime,
                LeadTimeDays = 0
            });
            
            remainingQty -= allocatedQty;
        }
        
        // 策略2：从在途库存分配
        if (remainingQty > 0)
        {
            var inTransits = context.InTransitInventories
                .Where(i => i.MaterialCode == order.MaterialCode && i.AvailableQty > 0)
                .OrderBy(i => i.EstimatedArrivalTime)
                .ToList();
            
            foreach (var inTransit in inTransits)
            {
                if (remainingQty <= 0) break;
                
                var allocatedQty = Math.Min(inTransit.AvailableQty, remainingQty);
                
                voucher.Details.Add(new PeggingDetail
                {
                    SourceType = "IN_TRANSIT",
                    SourceId = inTransit.Id,
                    SourceDescription = $"在途-{inTransit.MaterialCode}",
                    AllocatedQuantity = allocatedQty,
                    BalanceBefore = inTransit.AvailableQty,
                    BalanceAfter = inTransit.AvailableQty - allocatedQty,
                    AvailableTime = inTransit.EstimatedArrivalTime,
                    LeadTimeDays = (int)(inTransit.EstimatedArrivalTime - context.CurrentTime).TotalDays
                });
                
                remainingQty -= allocatedQty;
            }
        }
        
        // 如果仍有短缺，记录但不分配（等待计划生产）
        
        return voucher;
    }
    
    public PeggingVoucher SweepOrphans(ScheduleContext context)
    {
        // 扫描所有未完全满足的订单
        var orphanOrders = context.Orders
            .Where(o => !context.PeggingLedgers.ContainsKey(o.Id) || 
                       context.PeggingLedgers[o.Id].TotalShortageQty > 0)
            .ToList();
        
        // 为孤儿订单生成计划生产Pegging
        // 实现略...
        
        return new PeggingVoucher();
    }
}
```

### 5.4.1 接口定义

```csharp
/// 负责决定订单如何拆分成Task
/// </summary>
public interface ITaskSplitRule
{
    /// <summary>
    /// 计算订单的拆批策略
    /// </summary>
    /// <param name="order">订单</param>
    /// <param name="peggingLedger">Pegging账本</param>
    /// <returns>拆批凭证</returns>
    SplitVoucher CalculateSplit(Order order, PeggingLedger peggingLedger);
}
```

### 5.4.2 默认实现示例

```csharp
public class DefaultTaskSplitRule : ITaskSplitRule
{
    private const decimal MAX_BATCH_SIZE = 10000m; // 最大批次数量
    
    public SplitVoucher CalculateSplit(Order order, PeggingLedger peggingLedger)
    {
        var voucher = new SplitVoucher
        {
            OrderId = order.Id,
            OrderNo = order.OrderNo,
            CreatedBy = "DefaultTaskSplitRule"
        };
        
        // 策略1：小订单不拆批
        if (order.Quantity <= MAX_BATCH_SIZE)
        {
            voucher.SplitStrategy = "NO_SPLIT";
            voucher.SplitReason = "订单数量较小，不需要拆批";
            
            voucher.Instructions.Add(CreateTaskInstruction(order, 1, order.Quantity));
        }
        // 策略2：大订单按数量拆批
        else
        {
            voucher.SplitStrategy = "BY_QUANTITY";
            voucher.SplitReason = $"订单数量{order.Quantity}超过最大批次{MAX_BATCH_SIZE}，需要拆批";
            
            var batchCount = (int)Math.Ceiling(order.Quantity / MAX_BATCH_SIZE);
            for (int i = 0; i < batchCount; i++)
            {
                var batchQty = i == batchCount - 1 
                    ? order.Quantity - (i * MAX_BATCH_SIZE) 
                    : MAX_BATCH_SIZE;
                
                voucher.Instructions.Add(CreateTaskInstruction(order, i + 1, batchQty));
            }
        }
        
        return voucher;
    }
    
    private TaskGenerationInstruction CreateTaskInstruction(Order order, int splitIndex, decimal quantity)
    {
        return new TaskGenerationInstruction
        {
            TaskNo = $"T{order.OrderNo}_010_{splitIndex:D3}",
            OperationSeq = 10,
            OperationCode = "PRODUCTION",
            OperationName = "生产",
            Quantity = quantity,
            UOM = order.UOM,
            SplitIndex = splitIndex,
            PriorityScore = order.PriorityScore ?? 0,
            Duration = CalculateDuration(quantity),
            SetupTime = 0.5m,
            TaskType = "PRODUCTION",
            IsLocked = false,
            IsWip = false
        };
    }
    
    private decimal CalculateDuration(decimal quantity)
    {
        // 简化计算：假设每1000件需要1小时
        return quantity / 1000m;
    }
}
```

## 5.5 IFrozenZoneRule（冻结区判定插件）

### 5.5.1 接口定义

```csharp
/// <summary>
/// 冻结区判定插件接口
/// 负责判定哪些Task应该冻结（下发MES）
/// </summary>
public interface IFrozenZoneRule
{
    /// <summary>
    /// 评估Task是否应该冻结
    /// </summary>
    /// <param name="tasks">任务列表</param>
    /// <param name="currentTime">当前时间</param>
    /// <returns>冻结区打标凭证</returns>
    FrozenMarkVoucher Evaluate(List<Task> tasks, DateTime currentTime);
}
```

### 5.5.2 默认实现示例

```csharp
public class DefaultFrozenZoneRule : IFrozenZoneRule
{
    private const int FROZEN_WINDOW_DAYS = 7; // 冻结窗口：7天
    
    public FrozenMarkVoucher Evaluate(List<Task> tasks, DateTime currentTime)
    {
        var voucher = new FrozenMarkVoucher
        {
            FrozenCutoffTime = currentTime.AddDays(FROZEN_WINDOW_DAYS),
            CreatedBy = "DefaultFrozenZoneRule"
        };
        
        // 策略：计划开始时间在冻结窗口内的Task应该冻结
        var tasksToFreeze = tasks
            .Where(t => !t.IsFrozen && 
                       t.PlannedStartTime.HasValue &&
                       t.PlannedStartTime.Value <= voucher.FrozenCutoffTime)
            .ToList();
        
        foreach (var task in tasksToFreeze)
        {
            voucher.FrozenTasks.Add(new FrozenTaskMark
            {
                TaskId = task.Id,
                TaskNo = task.TaskNo,
                FrozenReason = $"计划开始时间{task.PlannedStartTime:yyyy-MM-dd}在冻结窗口内",
                ShouldRelease = true // 应该下发MES
            });
        }
        
        return voucher;
    }
}
```
