## 10.1 异常分类

/// <summary>

public abstract class BusinessException : Exception
{
    public string ErrorCode { get; set; }

    protected BusinessException(string errorCode, string message) : base(message)
    {
        ErrorCode = errorCode;

    }

    {
        Errors = errors;
    }

    }

## 10.2 异常处理策略

### 10.2.1 各号位异常处理原则

| 号位 | 异常处理策略 | 重试机制 | 回滚策略 |
|------|------------|---------|---------|
| **1号位** | 抛出异常，不处理 | 不重试 | 无需回滚（无状态） |
| **2号位** | 捕获并记录日志，事务回滚 | 数据库操作重试3次 | 事务自动回滚 |
| **3号位** | 捕获并记录，继续处理其他域 | 失败域重试1次 | 标记失败域 |
| **5号位** | 抛出异常，不处理 | 不重试 | 无需回滚（无状态） |

{

            {
                _logger.LogWarning($"凭证验证失败: {voucher.VoucherId}, 错误: {string.Join(", ", validation.Errors)}");
                throw new VoucherValidationException(validation.Errors);
            }
            
            // 2. 执行凭证
            // ... 执行逻辑
            
            _logger.LogInformation($"凭证执行成功: {voucher.GetDescription()}");
        }
        catch (VoucherValidationException ex)
        {
            // 业务异常：记录并重新抛出
            _logger.LogError(ex, $"凭证验证失败: {voucher.VoucherId}");
            throw;
        }
        catch (Exception ex)
        {
            // 系统异常：包装后抛出
            _logger.LogError(ex, $"凭证执行失败: {voucher.VoucherId}");
            throw new VoucherExecutionException($"凭证执行失败: {ex.Message}", voucher);
        }
    }
}

// 3号位异常处理示例
public class Orchestrator : IOrchestrator
{
    private readonly ILogger _logger;
    
    public async Task<OrchestratorResult> ExecuteFullScheduleAsync(int planVersionId)
    {
        var result = new OrchestratorResult
        {
            PlanVersionId = planVersionId,
            SuccessfulDomains = new List<string>(),
            FailedDomains = new Dictionary<string, string>()
        };
        
        var domains = await GetDomainsAsync(planVersionId);
        
        foreach (var domain in domains)
        {
            try
            {
                await ScheduleDomainAsync(domain, planVersionId);
                result.SuccessfulDomains.Add(domain);
                _logger.LogInformation($"域排产成功: {domain}");
            }
            catch (Exception ex)
            {
                // 记录失败，继续处理其他域
                result.FailedDomains[domain] = ex.Message;
                _logger.LogError(ex, $"域排产失败: {domain}");
            }
        }
        
        return result;
    }
}

## 10.3 重试机制

### 10.3.1 数据库操作重试

```csharp
public class RetryPolicy
{
    public static async Task<T> ExecuteWithRetryAsync<T>(
        Func<Task<T>> operation, 
        int maxRetries = 3, 
        int delayMs = 1000)
    {
        for (int i = 0; i < maxRetries; i++)
        {
            try
            {
                return await operation();
            }
            catch (Exception ex) when (i < maxRetries - 1 && IsTransientError(ex))
            {
                await Task.Delay(delayMs * (i + 1)); // 指数退避
            }
        }
        
        // 最后一次尝试，不捕获异常
        return await operation();
    }
    
    private static bool IsTransientError(Exception ex)
    {
        // 判断是否为临时性错误（如网络超时、死锁等）
        return ex is SqlException sqlEx && 
               (sqlEx.Number == -2 || sqlEx.Number == 1205);
    }
}

// 使用示例
var orders = await RetryPolicy.ExecuteWithRetryAsync(
    async () => await _dataLoader.LoadOrdersAsync(domainKey, planVersionId),
    maxRetries: 3,
    delayMs: 1000
);
```

## 10.4 日志规范

### 10.4.1 日志级别

| 级别 | 使用场景 | 示例 |
|------|---------|------|
| **TRACE** | 详细调试信息 | 凭证明细、循环迭代 |
| **DEBUG** | 调试信息 | 方法入口/出口、变量值 |
| **INFO** | 正常流程信息 | 排产开始/结束、凭证执行成功 |
| **WARN** | 警告信息 | 凭证验证失败、数据异常 |
| **ERROR** | 错误信息 | 异常捕获、操作失败 |
| **FATAL** | 致命错误 | 系统崩溃、数据损坏 |

### 11.3.2 监控代码示例

```csharp
public class PerformanceMonitor
{
    private readonly Stopwatch _stopwatch = new Stopwatch();
    private readonly PerformanceMetrics _metrics = new PerformanceMetrics();
    
    public void StartPhase(string phaseName)
    {
        _stopwatch.Restart();
    }
    
    public void EndPhase(string phaseName)
    {
        _stopwatch.Stop();
        
        switch (phaseName)
        {
            case "DataLoad":
                _metrics.DataLoadTimeMs = _stopwatch.ElapsedMilliseconds;
                break;
            case "VoucherExecution":
                _metrics.VoucherExecutionTimeMs = _stopwatch.ElapsedMilliseconds;
                break;
            case "Scheduling":
                _metrics.SchedulingTimeMs = _stopwatch.ElapsedMilliseconds;
                break;
            case "Persistence":
                _metrics.PersistenceTimeMs = _stopwatch.ElapsedMilliseconds;
                break;
        }
    }
    
    public PerformanceMetrics GetMetrics()
    {
        _metrics.TotalTimeMs = _metrics.DataLoadTimeMs + 
                               _metrics.VoucherExecutionTimeMs + 
                               _metrics.SchedulingTimeMs + 
                               _metrics.PersistenceTimeMs;
        
        _metrics.PeakMemoryMB = GC.GetTotalMemory(false) / 1024 / 1024;
        
        return _metrics;
    }
}

// 使用示例
var monitor = new PerformanceMonitor();

monitor.StartPhase("DataLoad");
var context = await _dataLoader.LoadDomainDataAsync(domainKey, planVersionId);
monitor.EndPhase("DataLoad");

// ... 其他阶段

var metrics = monitor.GetMetrics();
_logger.LogInformation(
    "排产完成: 总耗时={TotalTimeMs}ms, 订单数={OrderCount}, Task数={TaskCount}",
    metrics.TotalTimeMs,
    metrics.OrderCount,
    metrics.TaskCount
);
```
