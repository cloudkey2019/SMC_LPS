# 第三章：凭证（Voucher）契约规范

## 3.1 凭证设计原则

### 3.1.1 架构红线

**⚠️ 核心原则**：5号位绝对禁止直接修改数据，只能返回凭证；2号位根据凭证统一执行状态变更。

**设计理念**：
- **职责分离**：5号位负责"决策"，2号位负责"执行"
- **可追溯性**：所有状态变更都有凭证记录，便于审计和调试
- **可回滚性**：凭证可以被撤销或重做
- **可测试性**：5号位的业务逻辑可以独立测试，不依赖数据库

### 3.1.2 凭证基类

所有凭证必须继承自 `VoucherBase`：

```csharp
/// <summary>
/// 凭证基类
/// 所有凭证必须继承此类
/// </summary>
public abstract class VoucherBase
{
    /// <summary>
    /// 凭证ID（唯一标识）
    /// </summary>
    public Guid VoucherId { get; set; } = Guid.NewGuid();
    
    /// <summary>
    /// 凭证类型
    /// </summary>
    public abstract VoucherType Type { get; }
    
    /// <summary>
    /// 凭证版本号
    /// 用于版本兼容性控制
    /// </summary>
    public string Version { get; set; } = "1.0";
    
    /// <summary>
    /// 创建时间
    /// </summary>
    public DateTime CreatedAt { get; set; } = DateTime.Now;
    
    /// <summary>
    /// 创建者（5号位插件名称）
    /// </summary>
    public string CreatedBy { get; set; }
    
    /// <summary>
    /// 计划版本ID
    /// </summary>
    public int PlanVersionId { get; set; }
    
    /// <summary>
    /// 域标识
    /// </summary>
    public string DomainKey { get; set; }
    
    /// <summary>
    /// 验证凭证有效性
    /// 子类必须实现此方法
    /// </summary>
    public abstract ValidationResult Validate();
    
    /// <summary>
    /// 凭证描述（用于日志和调试）
    /// </summary>
    public abstract string GetDescription();
}

/// <summary>
/// 凭证类型枚举
/// </summary>
public enum VoucherType
{
    /// <summary>
    /// Pegging凭证（库存扣减）
    /// </summary>

    /// <summary>
    /// 拆批凭证（Task生成）
    /// </summary>

    /// <summary>
    /// 物料短缺凭证
    /// </summary>

    /// 冻结区打标凭证
    /// </summary>
    FrozenMark = 4,
    
    /// <summary>
    /// 厂间发货Task凭证
    /// </summary>
    ShippingTask = 5,
    
    /// <summary>
    /// 订单打分凭证
    /// </summary>
    OrderScore = 6
}

/// <summary>
/// 验证结果
/// </summary>
public class ValidationResult
{
    /// <summary>
    /// 是否验证通过
    /// </summary>
    public bool IsValid { get; set; }
    
    /// <summary>
    /// 错误消息列表
    /// </summary>
    public List<string> Errors { get; set; } = new List<string>();
    
    /// <summary>
    /// 警告消息列表
    /// </summary>
    public List<string> Warnings { get; set; } = new List<string>();
    
    /// <summary>
    /// 创建成功结果
    /// </summary>
    public static ValidationResult Success() => new ValidationResult { IsValid = true };
    
    /// <summary>
    /// 创建失败结果

    /// 创建失败结果（多个错误）
    /// </summary>

    { 
        IsValid = false, 
        Errors = errors 
    };
}
```

## 3.2 PeggingVoucher（库存扣减凭证）

### 3.2.1 业务用途

Pegging凭证记录了订单的物料需求如何从库存、在途或计划生产中得到满足。

### 3.2.2 完整定义

```csharp
/// <summary>
/// Pegging凭证（库存扣减凭证）
/// 由5号位IPeggingRule插件生成，2号位IVoucherExecutor执行
/// </summary>
public class PeggingVoucher : VoucherBase
{
    public override VoucherType Type => VoucherType.Pegging;
    
    // ============ 订单信息 ============
    
    /// <summary>
    /// 订单ID
    /// </summary>
    public long OrderId { get; set; }
    
    /// <summary>
    /// 订单编号（冗余字段，便于日志查看）
    /// </summary>
    public string OrderNo { get; set; }
    
    // ============ 物料信息 ============
    
    /// <summary>
    /// 物料编码
    /// </summary>
    public string MaterialCode { get; set; }
    
    /// <summary>
    /// 需求数量
    /// </summary>
    public decimal RequiredQuantity { get; set; }
    
    /// <summary>
    /// 计量单位
    /// </summary>
    public string UOM { get; set; }
    
    // ============ Pegging明细 ============
    
    /// <summary>
    /// Pegging明细列表
    /// 记录需求如何从各个供给源得到满足
    /// </summary>
    public List<PeggingDetail> Details { get; set; } = new List<PeggingDetail>();
    
    /// <summary>
    /// 总分配数量
    /// </summary>
    public decimal TotalAllocatedQuantity => Details.Sum(d => d.AllocatedQuantity);
    
    /// <summary>
    /// 短缺数量
    /// </summary>
    public decimal ShortageQuantity => RequiredQuantity - TotalAllocatedQuantity;
    
    /// <summary>
    /// 是否完全满足
    /// </summary>
    public bool IsFullySatisfied => ShortageQuantity <= 0;
    
    // ============ 验证方法 ============
    
    public override ValidationResult Validate()
    {
        var errors = new List<string>();
        
        if (OrderId <= 0)
            errors.Add("订单ID必须大于0");
            
        if (string.IsNullOrWhiteSpace(MaterialCode))
            errors.Add("物料编码不能为空");
            
        if (RequiredQuantity <= 0)
            errors.Add("需求数量必须大于0");
            
        if (Details == null || !Details.Any())
            errors.Add("Pegging明细不能为空");
            
        if (Details != null && Details.Any(d => d.AllocatedQuantity < 0))
            errors.Add("分配数量不能为负数");
            
        if (TotalAllocatedQuantity > RequiredQuantity)
            errors.Add($"总分配数量({TotalAllocatedQuantity})不能超过需求数量({RequiredQuantity})");
        
        return errors.Any() ? ValidationResult.Fail(errors) : ValidationResult.Success();
    }
    
    public override string GetDescription()
    {
        return $"Pegging凭证: 订单{OrderNo}, 物料{MaterialCode}, 需求{RequiredQuantity}, 分配{TotalAllocatedQuantity}, 短缺{ShortageQuantity}";
    }
}

/// <summary>
/// Pegging明细
/// </summary>
public class PeggingDetail
{
    /// <summary>
    /// 供给源类型
    /// INVENTORY（库存）/IN_TRANSIT（在途）/PLANNED_PRODUCTION（计划生产）
    /// </summary>
    public string SourceType { get; set; }
    
    /// <summary>
    /// 供给源ID
    /// - 如果是库存，则为InventoryRecord.Id
    /// - 如果是在途，则为InTransitInventory.Id
    /// - 如果是计划生产，则为上游Task.Id
    /// </summary>
    public long SourceId { get; set; }
    
    /// <summary>
    /// 供给源描述（便于日志查看）
    /// </summary>
    public string SourceDescription { get; set; }
    
    /// <summary>
    /// 分配数量
    /// </summary>
    public decimal AllocatedQuantity { get; set; }
    
    /// <summary>
    /// 扣减前余额
    /// </summary>
    public decimal BalanceBefore { get; set; }
    
    /// <summary>
    /// 扣减后余额
    /// </summary>
    public decimal BalanceAfter { get; set; }
    
    /// <summary>
    /// 可用时间（供给源的可用时间）
    /// </summary>
    public DateTime? AvailableTime { get; set; }
    
    /// <summary>
    /// 提前期（天）
    /// </summary>
    public int LeadTimeDays { get; set; }
}
```

### 3.2.3 使用示例

```csharp
// 5号位生成Pegging凭证
public class DefaultPeggingRule : IPeggingRule
{
    public PeggingVoucher CalculatePegging(Order order, ScheduleContext context)
    {
        var voucher = new PeggingVoucher
        {
            OrderId = order.Id,
            OrderNo = order.OrderNo,
            MaterialCode = order.MaterialCode,
            RequiredQuantity = order.Quantity,
            UOM = order.UOM,
            PlanVersionId = context.PlanVersionId,
            DomainKey = context.DomainKey,
            CreatedBy = "DefaultPeggingRule"
        };
        
        // 优先从库存分配
        var inventory = context.Inventories
            .FirstOrDefault(i => i.MaterialCode == order.MaterialCode && i.AvailableQty > 0);
        
        if (inventory != null)
        {
            var allocatedQty = Math.Min(inventory.AvailableQty, order.Quantity);
            voucher.Details.Add(new PeggingDetail
            {
                SourceType = "INVENTORY",
                SourceId = inventory.Id,
                SourceDescription = $"库存-{inventory.MaterialCode}",
                AllocatedQuantity = allocatedQty,
                BalanceBefore = inventory.AvailableQty,
                BalanceAfter = inventory.AvailableQty - allocatedQty,
                AvailableTime = DateTime.Now,
                LeadTimeDays = 0
            });
        }
        
        return voucher;
    }
}

// 2号位执行Pegging凭证
public class VoucherExecutor : IVoucherExecutor
{
    public void ExecutePeggingVoucher(PeggingVoucher voucher, ScheduleContext context)
    {
        // 1. 验证凭证
        var validation = voucher.Validate();
        if (!validation.IsValid)
            throw new VoucherValidationException(validation.Errors);
        
        // 2. 扣减库存
        foreach (var detail in voucher.Details)
        {
            if (detail.SourceType == "INVENTORY")
            {
                var inventory = context.Inventories.First(i => i.Id == detail.SourceId);
                inventory.AvailableQty -= detail.AllocatedQuantity;
            }
        }
        
        // 3. 创建Pegging连线
        foreach (var detail in voucher.Details)
        {
            context.PeggingLinks.Add(new PeggingLink
            {
                PlanVersionId = voucher.PlanVersionId,
                DownstreamTaskId = 0, // 此时Task还未生成
                DownstreamMaterialId = context.Materials.First(m => m.Code == voucher.MaterialCode).Id,
                SupplySourceType = detail.SourceType,
                SupplySourceId = detail.SourceId,
                Quantity = detail.AllocatedQuantity,
                LeadTimeDays = detail.LeadTimeDays
            });
        }
        
        // 4. 更新订单的Pegging账本
        var order = context.Orders.First(o => o.Id == voucher.OrderId);
        if (order.PeggingLedger == null)
            order.PeggingLedger = new PeggingLedger { OrderId = order.Id };
        
        // ... 更新账本逻辑
    }
}
```

## 3.3 SplitVoucher（拆批凭证）

### 3.3.1 业务用途

拆批凭证指示2号位如何将订单拆分成多个Task实例。

### 3.3.2 完整定义

```csharp
/// <summary>
/// 拆批凭证（Task生成凭证）
/// 由5号位ITaskSplitRule插件生成，2号位IVoucherExecutor执行
/// </summary>
public class SplitVoucher : VoucherBase
{
    public override VoucherType Type => VoucherType.Split;
    
    // ============ 订单信息 ============
    
    /// <summary>
    /// 订单ID
    /// </summary>
    public long OrderId { get; set; }
    
    /// <summary>
    /// 订单编号
    /// </summary>
    public string OrderNo { get; set; }
    
    // ============ 拆批策略 ============
    
    /// <summary>
    /// 拆批策略
    /// NO_SPLIT（不拆批）/BY_QUANTITY（按数量拆批）/BY_RESOURCE（按资源拆批）/BY_TIME（按时间拆批）
    /// </summary>
    public string SplitStrategy { get; set; }
    
    /// <summary>
    /// 拆批原因
    /// </summary>
    public string SplitReason { get; set; }
    
    // ============ Task生成指令 ============
    
    /// <summary>
    /// Task生成指令列表
    /// </summary>
    public List<TaskGenerationInstruction> Instructions { get; set; } = new List<TaskGenerationInstruction>();
    
    /// <summary>
    /// 总Task数量
    /// </summary>
    public int TotalTaskCount => Instructions.Count;
    
    // ============ 验证方法 ============
    
    public override ValidationResult Validate()
    {
        var errors = new List<string>();
        
        if (OrderId <= 0)
            errors.Add("订单ID必须大于0");
            
        if (Instructions == null || !Instructions.Any())
            errors.Add("Task生成指令不能为空");
            
        if (Instructions != null)
        {
            for (int i = 0; i < Instructions.Count; i++)
            {
                if (Instructions[i].Quantity <= 0)
                    errors.Add($"第{i+1}个Task的数量必须大于0");
                    
                if (Instructions[i].OperationSeq <= 0)
                    errors.Add($"第{i+1}个Task的工序序号必须大于0");
            }
        }
        
        return errors.Any() ? ValidationResult.Fail(errors) : ValidationResult.Success();
    }
    
    public override string GetDescription()
    {
        return $"拆批凭证: 订单{OrderNo}, 策略{SplitStrategy}, 生成{TotalTaskCount}个Task";
    }
}

/// <summary>
/// Task生成指令
/// </summary>
public class TaskGenerationInstruction
{
    /// <summary>
    /// Task编号
    /// 格式：T{OrderNo}_{OperationSeq}_{SplitIndex}
    /// </summary>
    public string TaskNo { get; set; }
    
    /// <summary>
    /// 工序序号
    /// </summary>
    public int OperationSeq { get; set; }
    
    /// <summary>
    /// 工序代码
    /// </summary>
    public string OperationCode { get; set; }
    
    /// <summary>
    /// 工序名称
    /// </summary>
    public string OperationName { get; set; }
    
    /// <summary>
    /// 大工艺阶段码 — BOM↔Routing 对接主键之二（v2.4 新增）
    /// ⚠️ 必须取自 StageDict（如 TJ_MACH / BJ_PAINT）；
    ///    决定 Task 落在哪个工厂的产能队列（工厂前缀）；
    ///    R20 跨组织场景 = 目标工厂视角（父件 TJ + 指派 BJ → BJ_MACH）。
    /// </summary>
    public string StageCode { get; set; }
    
    /// <summary>
    /// 任务数量
    /// </summary>
    public decimal Quantity { get; set; }
    
    /// <summary>
    /// 计量单位
    /// </summary>
    public string UOM { get; set; }
    
    /// <summary>
    /// 拆批索引（第几批）
    /// </summary>
    public int SplitIndex { get; set; }
    
    /// <summary>
    /// 工序编码（v5.0：替代原ResourceGroupId，通过OperationResourceEligibility动态确定可用资源）
    /// </summary>
    public string? OperationCode { get; set; }
    
    /// <summary>
    /// 优先级分数（继承自订单）
    /// </summary>
    public decimal PriorityScore { get; set; }
    
    /// <summary>
    /// 标准工期（小时）
    /// </summary>
    public decimal Duration { get; set; }
    
    /// <summary>
    /// 准备时间（小时）
    /// </summary>
    public decimal SetupTime { get; set; }
    
    /// <summary>
    /// 换型属性
    /// </summary>
    public string SetupAttribute { get; set; }
    
    /// <summary>
    /// 任务类型
    /// PRODUCTION/SETUP/SHIPPING
    /// </summary>
    public string TaskType { get; set; }
    
    /// <summary>
    /// 是否锁定
    /// </summary>
    public bool IsLocked { get; set; }
    
    /// <summary>
    /// 是否在制品
    /// </summary>
    public bool IsWip { get; set; }
}
```

### 3.3.3 使用示例

```csharp
// 5号位生成拆批凭证
public class DefaultTaskSplitRule : ITaskSplitRule
{
    public SplitVoucher CalculateSplit(Order order, PeggingLedger peggingLedger)
    {
        var voucher = new SplitVoucher
        {
            OrderId = order.Id,
            OrderNo = order.OrderNo,
            SplitStrategy = "NO_SPLIT",
            SplitReason = "订单数量较小，不需要拆批",
            CreatedBy = "DefaultTaskSplitRule"
        };
        
        // 简单策略：不拆批，生成一个Task
        voucher.Instructions.Add(new TaskGenerationInstruction
        {
            TaskNo = $"T{order.OrderNo}_010_001",
            OperationSeq = 10,
            OperationCode = "PRODUCTION",
            OperationName = "生产",
            Quantity = order.Quantity,
            UOM = order.UOM,
            SplitIndex = 1,
            PriorityScore = order.PriorityScore ?? 0,
            Duration = 8.0m, // 从工艺路线获取
            SetupTime = 0.5m,
            TaskType = "PRODUCTION",
            IsLocked = false,
            IsWip = false
        });
        
        return voucher;
    }
}

// 2号位执行拆批凭证
public void ExecuteSplitVoucher(SplitVoucher voucher, ScheduleContext context)
{

    foreach (var instruction in voucher.Instructions)
    {
        var task = new Task
        {
            TaskNo = instruction.TaskNo,
            PlanVersionId = voucher.PlanVersionId,

            MaterialId = order.MaterialId,

            OperationCode = instruction.OperationCode,

            Duration = instruction.Duration,
            SetupTime = instruction.SetupTime,

            IsLocked = instruction.IsLocked,
            IsWip = instruction.IsWip,
            IsFrozen = false,
            TaskType = instruction.TaskType,
            PriorityScore = instruction.PriorityScore,
            SplitIndex = instruction.SplitIndex,
            SetupAttribute = instruction.SetupAttribute,
            CreatedAt = DateTime.Now,
            UpdatedAt = DateTime.Now
        };
        
        context.Tasks.Add(task);
    }
}
```

```csharp
/// <summary>
```

```csharp
/// 由5号位IMaterialShortageRule插件生成

public class MaterialShortageVoucher : VoucherBase
```

## 3.8 凭证执行流程

### 3.8.1 标准执行流程

```csharp
public interface IVoucherExecutor
{
    /// <summary>
    /// 执行凭证的标准流程
    /// </summary>
    void Execute(VoucherBase voucher, ScheduleContext context)
    {
        // 1. 验证凭证
        var validation = voucher.Validate();
        if (!validation.IsValid)
        {
            throw new VoucherValidationException(
                $"凭证验证失败: {string.Join(", ", validation.Errors)}");
        }
        
        // 2. 记录日志
        context.Logs.Add(new ExecutionLog
        {
            Timestamp = DateTime.Now,
            Level = "INFO",
            Message = $"开始执行凭证: {voucher.GetDescription()}"
        });
        
        // 3. 执行凭证（根据类型分发）
        switch (voucher.Type)
        {
            case VoucherType.Pegging:
                ExecutePeggingVoucher((PeggingVoucher)voucher, context);
                break;
            case VoucherType.Split:
                ExecuteSplitVoucher((SplitVoucher)voucher, context);
                break;
            case VoucherType.OrderScore:
                ExecuteOrderScoreVoucher((OrderScoreVoucher)voucher, context);
                break;
            case VoucherType.FrozenMark:
                ExecuteFrozenMarkVoucher((FrozenMarkVoucher)voucher, context);
                break;
            case VoucherType.MaterialShortage:
                ExecuteMaterialShortageVoucher((MaterialShortageVoucher)voucher, context);
                break;
            case VoucherType.ShippingTask:
                ExecuteShippingTaskVoucher((ShippingTaskVoucher)voucher, context);
                break;
            default:
                throw new NotSupportedException($"不支持的凭证类型: {voucher.Type}");
        }
        
        // 4. 记录成功日志
        context.Logs.Add(new ExecutionLog
        {
            Timestamp = DateTime.Now,
            Level = "INFO",
            Message = $"凭证执行成功: {voucher.GetDescription()}"
        });
    }
}
```

```csharp
    }
}
```
