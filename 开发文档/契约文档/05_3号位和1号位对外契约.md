# 第六章：3号位对外契约（调度编排器）

## 6.1 IDomainScheduler（域调度接口）

```csharp
/// <summary>
/// 域调度接口
/// 负责单个域的排产调度
/// </summary>
public interface IDomainScheduler
{
    /// <summary>
    /// 执行单域排产
    /// </summary>
    /// <param name="domainKey">域标识</param>
    /// <param name="planVersionId">计划版本ID</param>
    /// <returns>排产结果</returns>
    Task<DomainScheduleResult> ScheduleDomainAsync(string domainKey, int planVersionId);
}
```

## 6.2 ITopologicalSorter（拓扑排序接口）

```csharp
/// <summary>
/// 拓扑排序接口
/// 负责对域进行拓扑排序，确定排产顺序
/// </summary>
public interface ITopologicalSorter
{
    /// <summary>
    /// 对域进行拓扑排序
    /// </summary>
    /// <param name="domains">域列表</param>
    /// <param name="dependencies">域依赖关系</param>
    /// <returns>排序后的域列表</returns>
    List<string> Sort(List<string> domains, Dictionary<string, List<string>> dependencies);
}
```

## 6.3 IOrchestrator（总编排器接口）

```csharp
/// <summary>
/// 总编排器接口
/// 负责协调所有域的排产
/// </summary>
public interface IOrchestrator
{
    /// <summary>
    /// 执行全量排产
    /// </summary>
    /// <param name="planVersionId">计划版本ID</param>
    /// <returns>排产结果</returns>
    Task<OrchestratorResult> ExecuteFullScheduleAsync(int planVersionId);
    
    /// <summary>
    /// 执行增量排产
    /// </summary>
    /// <param name="domainKey">域标识</param>
    /// <param name="planVersionId">计划版本ID</param>
    /// <returns>排产结果</returns>
    Task<OrchestratorResult> ExecuteIncrementalScheduleAsync(string domainKey, int planVersionId);
}
```

# 第七章：1号位对外契约（排程引擎）

## 7.1 ISchedulingEngine（排程引擎接口）

```csharp
/// <summary>
/// 排程引擎接口
/// 负责纯数学排程计算（排俄罗斯方块）
/// </summary>
public interface ISchedulingEngine
{
    /// <summary>
    /// 执行排程计算
    /// </summary>
    /// <param name="input">排程输入</param>
    /// <returns>排程输出</returns>
    Task<SchedulingOutput> ScheduleAsync(SchedulingInput input);
}
```

## 7.2 SchedulingInput（排程输入）

```csharp
/// <summary>
/// 排程输入数据结构
/// </summary>
public class SchedulingInput
{
    /// <summary>
    /// 待排程任务列表
    /// </summary>
    public List<Task> Tasks { get; set; }
    
    /// <summary>
    /// 资源日历
    /// </summary>
    public ResourceCalendar ResourceCalendar { get; set; }
    
    /// <summary>
    /// 历史锚点（已冻结任务）
    /// </summary>
    public List<Task> HistoricalAnchors { get; set; }
    
    /// <summary>
    /// 排程参数
    /// </summary>
    public SchedulingParameters Parameters { get; set; }
}

/// <summary>
/// 排程参数
/// </summary>
public class SchedulingParameters
{
    /// <summary>
    /// 排程算法类型
    /// FIFO/PRIORITY/GENETIC/SIMULATED_ANNEALING
    /// </summary>
    public string AlgorithmType { get; set; }
    
    /// <summary>
    /// 最大迭代次数
    /// </summary>
    public int MaxIterations { get; set; }
    
    /// <summary>
    /// 超时时间（秒）
    /// </summary>
    public int TimeoutSeconds { get; set; }
    
    /// <summary>
    /// 是否考虑换型时间
    /// </summary>
    public bool ConsiderSetupTime { get; set; }
}
```

## 7.3 SchedulingOutput（排程输出）

```csharp
/// <summary>
/// 排程输出数据结构
/// </summary>
public class SchedulingOutput
{
    /// <summary>
    /// 任务排程结果列表
    /// </summary>
    public List<TaskScheduleResult> TaskResults { get; set; }
    
    /// <summary>
    /// 资源利用率
    /// </summary>
    public Dictionary<int, decimal> ResourceUtilization { get; set; }
    
    /// <summary>
    /// 排程质量指标
    /// </summary>
    public SchedulingMetrics Metrics { get; set; }
    
    /// <summary>
    /// 执行时间（毫秒）
    /// </summary>
    public long ExecutionTimeMs { get; set; }
}

/// <summary>
/// 任务排程结果
/// </summary>
public class TaskScheduleResult
{
    /// <summary>
    /// 任务ID
    /// </summary>
    public long TaskId { get; set; }
    
    /// <summary>
    /// 分配的资源ID
    /// </summary>
    public int ResourceId { get; set; }
    
    /// <summary>
    /// 计划开始时间
    /// </summary>
    public DateTime StartTime { get; set; }
    
    /// <summary>
    /// 计划结束时间
    /// </summary>
    public DateTime EndTime { get; set; }
    
    /// <summary>
    /// 是否关键路径
    /// </summary>
    public bool IsCriticalPath { get; set; }
}

/// <summary>
/// 排程质量指标
/// </summary>
public class SchedulingMetrics
{
    /// <summary>
    /// 平均资源利用率
    /// </summary>
    public decimal AverageUtilization { get; set; }
    
    /// <summary>
    /// 延期订单数量
    /// </summary>
    public int DelayedOrderCount { get; set; }
    
    /// <summary>
    /// 平均延期天数
    /// </summary>
    public decimal AverageDelayDays { get; set; }
    
    /// <summary>
    /// 换型次数
    /// </summary>
    public int SetupCount { get; set; }
}
```
